local function logoImage()
	local ok, res = pcall(function()
		local obj = game:GetObjects("rbxassetid://" .. LOGO_ID)[1]
		if obj and (obj:IsA("Decal") or obj:IsA("Texture")) then return obj.Texture end
		if obj and obj:IsA("ImageLabel") then return obj.Image end
	end)
	if ok and res and res ~= "" then return res end
	return "rbxassetid://" .. LOGO_ID
end

-- ====== PASANG UI DI LUAR MENU ROBLOX (tetap tampil saat ESC dibuka) ======
-- Urutan: gethui() -> CoreGui -> PlayerGui (cadangan kalau executor tidak mendukung)
local function mountGui(g)
	g.ResetOnSpawn = false
	g.DisplayOrder = 999
	g.IgnoreGuiInset = true                                       -- boleh sampai ke area topbar Roblox
	pcall(function() g.ScreenInsets = Enum.ScreenInsets.None end) -- tanpa batas safe-area
	pcall(function() g.ZIndexBehavior = Enum.ZIndexBehavior.Sibling end)
	local ok = false
	if typeof(gethui) == "function" then
		ok = pcall(function() g.Parent = gethui() end) and g.Parent ~= nil
	end
	if not ok then
		ok = pcall(function() g.Parent = game:GetService("CoreGui") end) and g.Parent ~= nil
	end
	if not ok then
		g.Parent = game:GetService("Players").LocalPlayer:WaitForChild("PlayerGui")
	end
	return g
end

	local function addPress(button)
		local sc = make("UIScale", {Scale = 1}, button)
		local function release() tween(sc, 0.3, {Scale = 1}, Enum.EasingStyle.Back) end
		button.MouseButton1Down:Connect(function() tween(sc, 0.1, {Scale = 0.95}) end)
		button.MouseButton1Up:Connect(release)
		button.MouseLeave:Connect(release)
	end

	local function addHover(button, normal, hover)
		button.MouseEnter:Connect(function() tween(button, 0.18, {BackgroundColor3 = hover}) end)
		button.MouseLeave:Connect(function() tween(button, 0.25, {BackgroundColor3 = normal}) end)
	end

	-- ====== HELPER ======
	local function getPos(target)
		if target:IsA("Model") then
			return target:GetPivot().Position
		end
		return target.Position
	end

	local function firstPart(target)
		if target:IsA("BasePart") then return target end
		return target:FindFirstChildWhichIsA("BasePart", true)
	end

	local function getParts(obj)
		local parts = {}
		if obj:IsA("BasePart") then
			table.insert(parts, obj)
		elseif obj:IsA("Model") then
			for _, item in ipairs(obj:GetDescendants()) do
				if item:IsA("BasePart") then
					table.insert(parts, item)
				end
			end
		end
		return parts
	end

	local function pumpkinVisible(obj)
		if not obj:IsDescendantOf(workspace) then
			return false
		end
		for _, part in ipairs(getParts(obj)) do
			if part.Transparency < 1
				and part.LocalTransparencyModifier < 1
				and part.Size.Magnitude > 0 then
				return true
			end
		end
		return false
	end

	local function insideOffering(obj)
		local p = obj.Parent
		while p and p ~= workspace do
			if p:IsA("Model") and p.Name:match(OFFERING_PREFIX) then
				return true
			end
			p = p.Parent
		end
		return false
	end

	local function insideCandidate(obj)
		local p = obj.Parent
		while p and p ~= workspace do
			if candidates[p] then return true end
			p = p.Parent
		end
		return false
	end

	local function offeringDisplayName(name)
		if OFFERING_NAMES[name] then
			return OFFERING_NAMES[name]
		end
		local short = (name:gsub(OFFERING_PREFIX, ""))
		if short == "" then return name end
		return short
	end

	-- ====== LABEL PINTU ======
	local function relabel(quiet)
		local list = {}
		for _, c in pairs(candidates) do
			if c.isFF then table.insert(list, c) end
		end

		table.sort(list, function(a, b)
			local va, vb = getPos(a.target)[ORDER_BY], getPos(b.target)[ORDER_BY]
			if va ~= vb then return va < vb end
			return a.target:GetFullName() < b.target:GetFullName()
		end)

		for i, c in ipairs(list) do
			local lab = LABELS[i]
			if lab then
				c.name = lab[1]
				c.nameEN = L.LABEL_EN[lab[1]]
				c.color = lab[2] or DEFAULT_FF_COLOR
			else
				c.name = "Forcefield #" .. i
				c.nameEN = nil
				c.color = DEFAULT_FF_COLOR
			end
		end

		if DEBUG and not quiet then
			print("[ESP] Urutan Forcefield (berdasarkan " .. ORDER_BY .. "):")
			for i, c in ipairs(list) do
				local p = getPos(c.target)
				print(string.format("  #%d  %s  | X=%d Y=%d Z=%d | %s (%s)",
					i, c.name, p.X, p.Y, p.Z, c.target:GetFullName(), c.target.ClassName))
			end
		end
	end

		candidates[target] = {
			target = target, adornee = firstPart(target),
			color = MONSTER_COLOR, name = MONSTER_NAMES[target.Name] or target.Name,
			isFF = false, kind = "mon",
		}
	end

	local function register(obj)
		-- 1) Model Offering
		if obj:IsA("Model") and obj.Name:match(OFFERING_PREFIX) and not insideOffering(obj) then
			if not candidates[obj] then
				candidates[obj] = {
					target = obj, adornee = firstPart(obj),
					color = OFFERING_COLOR, name = offeringDisplayName(obj.Name), isFF = false, kind = "off",
				}
			end
			return false
		end

		-- 2) Forcefield (Model atau Part)
		if obj.Name:lower() == FF_NAME
			and (obj:IsA("Model") or obj:IsA("BasePart"))
			and not candidates[obj]
			and not insideOffering(obj)
			and not insideCandidate(obj) then
			candidates[obj] = {
				target = obj, adornee = firstPart(obj),
				color = DEFAULT_FF_COLOR, name = "Forcefield", isFF = true, kind = "ff",
			}
			return true
		end

		-- 3) Pumpkin (Model atau Part)
		if obj.Name:lower() == PUMPKIN_NAME:lower()
			and (obj:IsA("Model") or obj:IsA("BasePart"))
			and not candidates[obj]
			and not insideOffering(obj)
			and not insideCandidate(obj) then
			candidates[obj] = {
				target = obj, adornee = firstPart(obj),
				color = PUMPKIN_COLOR, outline = PUMPKIN_OUTLINE,
				name = PUMPKIN_LABEL, isFF = false, kind = "pump",
			}
			return false
		end

		-- 3b) Brazier (tempat ambil torch)
		if obj.Name:lower() == BRAZIER_NAME:lower()
			and (obj:IsA("Model") or obj:IsA("BasePart"))
			and not candidates[obj]
			and not insideOffering(obj)
			and not insideCandidate(obj) then
			candidates[obj] = {
				target = obj, adornee = firstPart(obj),
				color = BRAZIER_COLOR, name = BRAZIER_LABEL, isFF = false, kind = "brz",
			}
			return false
		end

		-- 3b2) PuzzleLever (floor 9): Model/Part bernama PuzzleLever
		if obj.Name:lower() == LEVER_NAME:lower()
			and (obj:IsA("Model") or obj:IsA("BasePart"))
			and not candidates[obj]
			and not insideOffering(obj)
			and not insideCandidate(obj) then
			candidates[obj] = {
				target = obj, adornee = firstPart(obj),
				color = LEVER_COLOR, name = LEVER_LABEL, nameEN = "LEVER", isFF = false, kind = "lever",
			}
			return false
		end

		-- 3c) Toko: Cylinder.009 di dalam folder Poles
		if obj.Name == SHOP_PART
			and (obj:IsA("BasePart") or obj:IsA("Model"))
			and obj.Parent and obj.Parent.Name:lower() == SHOP_FOLDER:lower()
			and not candidates[obj] then
			candidates[obj] = {
				target = obj, adornee = firstPart(obj),
				color = SHOP_COLOR, name = SHOP_LABEL, nameEN = "SHOP", isFF = false, kind = "shop",
			}
			return false
		end

		-- 4) Monster: model yang langsung ada di folder "Mobs"
		if obj:IsA("Model") and obj.Parent and obj.Parent.Name:lower() == MOBS_FOLDER:lower() then
			addMonster(obj)
			return false
		end

		return false
	end

	local function buildCandidates(quiet)
		-- kalau sudah ada scan yang jalan, tunggu saja sampai selesai
		if L.building then
			while L.building do task.wait() end
			return
		end
		L.building = true
		candidates = {}
		local mirip = {}
		local cnt = 0
		for _, obj in ipairs(workspace:GetDescendants()) do
			pcall(register, obj)
			if #mirip < 15 and obj.Name:lower():find("force") then
				table.insert(mirip, obj:GetFullName() .. " [" .. obj.ClassName .. "]")
			end
			cnt += 1
			if cnt % 1200 == 0 then task.wait() end   -- bagi kerja ke beberapa frame, biar UI tidak nyangkut
		end
		L.building = false
		L.builtOnce = true
		relabel(quiet)
		if DEBUG and not quiet then
			local n = {mon = 0, ff = 0, off = 0, pump = 0, brz = 0, shop = 0, lever = 0}
			local pumpNongol = 0
			for _, c in pairs(candidates) do
				n[c.kind] += 1
				if c.kind == "pump" and pumpkinVisible(c.target) then
					pumpNongol += 1
				end
			end
			print("[ESP] Ditemukan -> monster:", n.mon, "| pintu:", n.ff, "| offering:", n.off,
				"| pumpkin total:", n.pump, "(nongol:", pumpNongol .. ")", "| tuas:", n.lever)
			if n.mon == 0 then
				print("[ESP] Monster 0. Pastikan folder '" .. MOBS_FOLDER .. "' ada di Workspace (bukan ServerStorage/ReplicatedStorage).")
			end
			if n.pump == 0 then
				print("[ESP] Pumpkin 0. Kalau di floor 7, pastikan namanya '" .. PUMPKIN_NAME .. "' dan sudah dimuat (dekati areanya).")
			end
			if n.lever == 0 then
				print("[ESP] Tuas 0. Kalau di floor 9, pastikan namanya '" .. LEVER_NAME .. "' dan sudah dimuat (dekati areanya).")
			end
			print("[ESP] Objek yang namanya mengandung 'force':")
			for _, s in ipairs(mirip) do print("   ", s) end
			if #mirip == 0 then
				print("   (tidak ada. Mungkin StreamingEnabled / objek belum dimuat, dekati pintunya dulu)")
			end
		end
	end

	local function removeItem(key, instant)
		local it = items[key]
		if not it then return end
		items[key] = nil
		if it.box then it.box:Destroy() end
		if instant then
			it.hl:Destroy()
			it.bb:Destroy()
			return
		end
		tween(it.hl, 0.2, {FillTransparency = 1, OutlineTransparency = 1})
		tween(it.pill, 0.2, {BackgroundTransparency = 1})
		tween(it.stroke, 0.2, {Transparency = 1})
		tween(it.label, 0.2, {TextTransparency = 1})
		task.delay(0.25, function()
			it.hl:Destroy()
			it.bb:Destroy()
		end)
	end

	local function clearAll()
		for key in pairs(items) do
			removeItem(key, true)
		end
	end

	local function setStat(kind, value)
		local s = statBoxes[kind]
		if not s then return end
		local txt = tostring(value)
		if s.num.Text ~= txt then
			s.num.Text = txt
			if value ~= "–" then
				s.scale.Scale = 1.35
				tween(s.scale, 0.35, {Scale = 1}, Enum.EasingStyle.Back)
			end
		end
	end

	local function getRoot()
		local ch = player.Character
		local hrp = ch and ch:FindFirstChild("HumanoidRootPart")
		return hrp, ch
	end

	local function getBounds(target)
		if target:IsA("Model") then
			return target:GetBoundingBox()
		end
		return target.CFrame, target.Size
	end

	-- teleport ke dekat target (berdiri di lantai, menghadap target)
	local function teleportTo(c)
		local hrp, ch = getRoot()
		if not hrp or not c.target:IsDescendantOf(workspace) then return false end
		local cf, size = getBounds(c.target)
		local center = cf.Position
		local floorY = center.Y - size.Y / 2
		local flat = Vector3.new(hrp.Position.X - center.X, 0, hrp.Position.Z - center.Z)
		local dir = flat.Magnitude > 0.5 and flat.Unit or Vector3.new(0, 0, 1)
		local radius = math.clamp(math.max(size.X, size.Z) / 2 + 4, 5, 16)
		local dest = Vector3.new(center.X, floorY + 3, center.Z) + dir * radius
		if c.kind == "shop" and SHOP_OFFSET then
			local pv = c.target:IsA("Model") and c.target:GetPivot() or c.target.CFrame
			dest = pv:PointToWorldSpace(SHOP_OFFSET)
		end
		ch:PivotTo(CFrame.lookAt(dest, Vector3.new(center.X, dest.Y, center.Z)))
		hrp.AssemblyLinearVelocity = Vector3.zero
		return true
	end

	local function promptPos(p)
		local par = p.Parent
		if par and par:IsA("BasePart") then return par.Position end
		if par and par:IsA("Attachment") then return par.WorldPosition end
		return nil
	end

	-- cari ProximityPrompt di dalam target (deep = cari juga prompt di sekitar target)
	local function findPrompts(target, match, deep)
		local list = {}
		for _, d in ipairs(target:GetDescendants()) do
			if d:IsA("ProximityPrompt") and d.Enabled then table.insert(list, d) end
		end
		if #list == 0 and deep then
			local cf = getBounds(target)
			for _, d in ipairs(workspace:GetDescendants()) do
				if d:IsA("ProximityPrompt") and d.Enabled then
					local pp = promptPos(d)
					if pp and (pp - cf.Position).Magnitude <= 15 then table.insert(list, d) end
				end
			end
		end
		if match then
			local filtered = {}
			for _, p in ipairs(list) do
				if match(p) then table.insert(filtered, p) end
			end
			if #filtered > 0 then return filtered end
			if #list == 1 then return list end
			return {}
		end
		return list
	end

	local function triggerPrompt(prompt)
		if typeof(fireproximityprompt) == "function" then
			pcall(function() prompt.HoldDuration = 0 end)
			local ok = pcall(fireproximityprompt, prompt)
			if ok then return true end
		end
		-- cadangan: tekan tombol prompt lewat VirtualInputManager
		local ok = pcall(function()
			local vim = game:GetService("VirtualInputManager")
			vim:SendKeyEvent(true, prompt.KeyboardKeyCode, false, game)
			task.wait(math.max(prompt.HoldDuration, 0.05) + 0.1)
			vim:SendKeyEvent(false, prompt.KeyboardKeyCode, false, game)
		end)
		return ok
	end

	local function matchTorch(p) return ((p.ActionText or "") .. " " .. (p.ObjectText or "")):lower():find("torch") ~= nil end
	local function matchLight(p) return (p.ActionText or ""):lower():find("light") ~= nil end

	local function torchTool()
		local ch = player.Character
		if ch then
			for _, t in ipairs(ch:GetChildren()) do
				if t:IsA("Tool") and t.Name:lower():find("torch") then return t, true end
			end
		end
		local bp = player:FindFirstChildOfClass("Backpack")
		if bp then
			for _, t in ipairs(bp:GetChildren()) do
				if t:IsA("Tool") and t.Name:lower():find("torch") then return t, false end
			end
		end
		return nil
	end

	local function equipTorch()
		local tool, equipped = torchTool()
		if tool and not equipped then
			local ch = player.Character
			local hum = ch and ch:FindFirstChildOfClass("Humanoid")
			if hum then pcall(function() hum:EquipTool(tool) end) end
		end
		return tool ~= nil
	end

	local function interact(c, deep)
		-- tuas (lever) sengaja TIDAK ditarik otomatis; teleport saja
		if c.kind == "ff" or c.kind == "shop" or c.kind == "lever" then return 0 end
		local match = (c.kind == "brz" and matchTorch) or (c.kind == "pump" and matchLight) or nil
		local n = 0
		for _, p in ipairs(findPrompts(c.target, match, deep)) do
			if triggerPrompt(p) then n += 1 end
			task.wait(0.1)
		end
		return n
	end

	-- daftar target (sudah terurut) untuk satu jenis
	local function collectTargets(kind)
		local hrp = getRoot()
		local out = {}
		for _, c in pairs(candidates) do
			if c.kind == kind and c.target:IsDescendantOf(workspace) then
				if kind ~= "pump" or pumpkinVisible(c.target) then
					local d = hrp and (getPos(c.target) - hrp.Position).Magnitude or 0
					table.insert(out, {c = c, dist = d})
				end
			end
		end
		if kind == "ff" then
			table.sort(out, function(a, b)
				local va, vb = getPos(a.c.target)[ORDER_BY], getPos(b.c.target)[ORDER_BY]
				if va ~= vb then return va < vb end
				return a.c.target:GetFullName() < b.c.target:GetFullName()
			end)
		else
			table.sort(out, function(a, b) return a.dist < b.dist end)
		end
		return out
	end

	-- klik tombol Teleport di daftar
	local function goTo(c, displayName)
		if not teleportTo(c) then
			showToast('<font color="#8E8E98">●</font>  ' .. esc(L.T("toast_lost")), THEME.off)
			return
		end
		showToast('<font color="' .. hex(c.color) .. '">●</font>  ' .. esc(L.T("toast_tp", displayName)), c.color)
		if AUTO_PROMPT and c.kind ~= "ff" and c.kind ~= "shop" and c.kind ~= "lever" then
			task.wait(0.35)
			if c.kind == "pump" then equipTorch() end
			local n = interact(c, true)
			if n > 0 then
				task.wait(0.9)
				showToast('<font color="#46DC8C">●</font>  ' .. esc(L.T("toast_prompt", displayName)), THEME.on)
			end
		end
	end

	-- ===== AUTO: Get Torch lalu nyalakan Pumpkin (floor 7) =====
	local function stopAuto(msg)
		autoToken += 1
		autoRunning = false
		setAutoStatus(msg or "auto_stopped")
		if autoSwitchRef then pcall(autoSwitchRef.set, false) end
	end

	local function claimTorch()
		local list = collectTargets("brz")
		if #list == 0 then return false, "auto_nobrz" end
		local c = list[1].c
		teleportTo(c)
		task.wait(0.4)
		for _ = 1, 3 do
			for _, p in ipairs(findPrompts(c.target, matchTorch, true)) do
				triggerPrompt(p)
				task.wait(0.1)
			end
			task.wait(0.7)
			if torchTool() then equipTorch() break end
		end
		return true
	end

	local function startAuto()
		if autoRunning then return end
		autoToken += 1
		local myToken = autoToken
		autoRunning = true
		setAutoStatus("auto_start")
		task.spawn(function()
			local floorAtStart = currentFloor
			buildCandidates(true)
			local lastBuild = os.clock()
			local claimed, torchSeen = false, false
			local attempts = {}
			local function alive() return autoRunning and myToken == autoToken and currentFloor == floorAtStart and gui.Parent end

			while alive() do
				if os.clock() - lastBuild > 4 then
					buildCandidates(true)
					lastBuild = os.clock()
				end

				local targets = {}
				for _, t in ipairs(collectTargets("pump")) do
					if (attempts[t.c.target] or 0) < 3 and #findPrompts(t.c.target, matchLight, false) > 0 then
						table.insert(targets, t)
					end
				end

				if #targets == 0 then
					setAutoStatus("auto_wait")
					task.wait(1)
				else
					local tool = torchTool()
					if tool then torchSeen = true end
					if not claimed or (torchSeen and not tool) then
						setAutoStatus("auto_torch")
						claimed = true
						local ok, note = claimTorch()
						if not ok then
							if myToken == autoToken then stopAuto(note) end
							return
						end
					end
					equipTorch()
					for i, t in ipairs(targets) do
						if not alive() then break end
						setAutoStatus("auto_light", i, #targets)
						if teleportTo(t.c) then
							task.wait(0.3)
							attempts[t.c.target] = (attempts[t.c.target] or 0) + 1
							interact(t.c, false)
							task.wait(0.45)
						end
					end
				end
			end
			if myToken == autoToken then stopAuto("auto_stopped") end
		end)
	end

	-- ====== SCAN ======
	local function scan()
		if L.building then return end
		local char = player.Character
		local hrp = char and char:FindFirstChild("HumanoidRootPart")
		if not hrp then return end

		local cfg = FLOORS[currentFloor]
		local found = {}
		local count = {ff = 0, off = 0, mon = 0, pump = 0, brz = 0, shop = 0, lever = 0}

		for key, c in pairs(candidates) do
			if not c.target:IsDescendantOf(workspace) then
				candidates[key] = nil
				removeItem(key, true)
			elseif cfg[c.kind] and (c.kind ~= "shop" or SHOP_ESP) then
				if c.kind == "pump" and not pumpkinVisible(c.target) then
					-- tidak dimasukkan ke found, jadi ESP-nya otomatis dicabut
				else
					if not c.adornee or not c.adornee:IsDescendantOf(workspace) then
						c.adornee = firstPart(c.target)
					end
					local dist = (getPos(c.target) - hrp.Position).Magnitude
					if dist <= RADIUS and c.adornee then
						table.insert(found, {key = key, c = c, dist = dist})
						count[c.kind] += 1
					end
				end
			end
		end

		table.sort(found, function(a, b) return a.dist < b.dist end)

		local keep = {}
		for i = 1, math.min(#found, MAX_ESP) do
			local f = found[i]
			keep[f.key] = true
			if not items[f.key] then addItem(f.key, f.c) end

			local it = items[f.key]
			local txt = labelText(f.c, math.floor(f.dist))
			if it.text ~= txt then
				it.text = txt
				it.label.Text = txt
			end
			it.stroke.Color = f.c.color
			it.hl.FillColor = f.c.color
			if it.box then it.box.Color3 = f.c.color end
		end

		for key in pairs(items) do
			if not keep[key] then removeItem(key) end
		end

		updateStats(count)
	end

	local function setESP(state)
		espOn = state
		runId += 1
		local myId = runId
		if espOn then
			addedConn = workspace.DescendantAdded:Connect(function(obj)
				task.wait()
				if register(obj) then relabel() end
			end)
			task.spawn(function()
				buildCandidates()
				while espOn and myId == runId do
					scan()
					task.wait(0.4)
				end
			end)
		else
			if addedConn then addedConn:Disconnect() addedConn = nil end
			clearAll()
		end
	end

	local function toggle()
		setESP(not espOn)
		applyToggleVisual()
		if espOn then
			showToast('<font color="#46DC8C">●</font>  ' .. esc(L.T("toast_esp_on", currentFloor)), THEME.on)
		else
			updateStats({mon = 0, ff = 0, off = 0, pump = 0})
			showToast('<font color="#8E8E98">●</font>  ' .. esc(L.T("toast_esp_off")), THEME.off)
		end
	end

	local function selectFloor(n)

		refreshFloorButtons()
		clearAll()
		refreshStatsActive()
		refreshHeader()
		if espOn then scan() end

	end

	function L.flyStop()
		local f = L.fly
		if f.conn then f.conn:Disconnect() f.conn = nil end
		if f.bv then f.bv:Destroy() f.bv = nil end
		if f.bg then f.bg:Destroy() f.bg = nil end
		f.up, f.down = false, false
		local _, hrp, hum = L.charParts()
		if hum then hum.PlatformStand = false end
		if hrp then hrp.AssemblyLinearVelocity = Vector3.zero end
	end

	function L.flyStart()
		L.flyStop()
		local _, hrp, hum = L.charParts()
		if not (hrp and hum) then return false end
		local f = L.fly
		local bv = Instance.new("BodyVelocity")
		bv.MaxForce = Vector3.new(9e9, 9e9, 9e9)
		bv.Velocity = Vector3.zero
		bv.Parent = hrp
		local bg = Instance.new("BodyGyro")
		bg.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
		bg.P = 9e4
		bg.D = 1000
		bg.CFrame = hrp.CFrame
		bg.Parent = hrp
		f.bv, f.bg = bv, bg
		hum.PlatformStand = true
		local cur = Vector3.zero
		f.conn = game:GetService("RunService").RenderStepped:Connect(function(dt)
			local cam = workspace.CurrentCamera
			if not cam or not hrp.Parent or not hum.Parent then return end
			local look = cam.CFrame.LookVector
			local flat = Vector3.new(look.X, 0, look.Z)
			flat = flat.Magnitude > 0.01 and flat.Unit or Vector3.new(0, 0, -1)
			local move = hum.MoveDirection              -- sudah relatif kamera (keyboard & joystick HP)
			local target = move * f.speed
			if move.Magnitude > 0.05 then               -- arah kamera naik/turun ikut menentukan ketinggian
				target = target + Vector3.new(0, look.Y * move:Dot(flat) * f.speed, 0)
			end
			local vy = 0
			if UIS:IsKeyDown(Enum.KeyCode.Space) or f.up then vy += 1 end
			if UIS:IsKeyDown(Enum.KeyCode.LeftShift) or UIS:IsKeyDown(Enum.KeyCode.LeftControl) or f.down then vy -= 1 end
			target = target + Vector3.new(0, vy * f.speed, 0)
			cur = cur:Lerp(target, math.clamp(dt * 10, 0, 1))   -- akselerasi halus
			bv.Velocity = cur
			bg.CFrame = CFrame.lookAt(hrp.Position, hrp.Position + flat)
		end)
		return true
	end

	-- animasi ganti bahasa: HANYA teks yang memudar & muncul lagi (latar tetap), dengan efek gelombang atas -> bawah
	function L.animateSwap()
		local function collect()
			local out = {}
			for _, d in ipairs(card:GetDescendants()) do
				if (d:IsA("TextLabel") or d:IsA("TextButton")) and d.Text ~= "" and d.Text:find("%a")
					and not d.Text:find("^The Hollow") and not d:IsDescendantOf(logo) then
					local hidden = false
					for _, pgx in ipairs(pages) do
						if not pgx.Visible and d:IsDescendantOf(pgx) then hidden = true break end
					end
					if not hidden then table.insert(out, d) end
				end
			end
			return out
		end
		local old = collect()
		local goal = {}
		for _, d in ipairs(old) do
			goal[d] = d.TextTransparency
			tween(d, 0.12, {TextTransparency = 1}, Enum.EasingStyle.Quad)
		end
		task.delay(0.13, function()
			L.apply()                                   -- teks diganti saat sedang tak terlihat
			local seen = {}
			local top, hgt = card.AbsolutePosition.Y, math.max(1, card.AbsoluteSize.Y)
			for _, d in ipairs(collect()) do
				seen[d] = true
				local g = goal[d] or d.TextTransparency
				d.TextTransparency = 1
				local wave = math.clamp((d.AbsolutePosition.Y - top) / hgt, 0, 1) * 0.2
				task.delay(wave, function()
					if d.Parent then tween(d, 0.32, {TextTransparency = g}, Enum.EasingStyle.Quint) end
				end)
			end
			for _, d in ipairs(old) do                  -- yang sempat disembunyikan tapi tak ikut lagi: kembalikan
				if not seen[d] and d.Parent then d.TextTransparency = goal[d] end
			end
		end)
	end

	-- ====== DRAG (PC & HP) ======
	local function makeDraggable(handle, target, onTap, threshold, onMove)
		threshold = threshold or 6
		local dragging, moved = false, false
		local dragStart, startPos

		handle.InputBegan:Connect(function(input)
			if input.UserInputType == Enum.UserInputType.MouseButton1
				or input.UserInputType == Enum.UserInputType.Touch then
				dragging = true
				moved = false
				dragStart = input.Position
				startPos = target.Position
			end
		end)

		track(UIS.InputChanged:Connect(function(input)
			if not dragging then return end
			if input.UserInputType == Enum.UserInputType.MouseMovement
				or input.UserInputType == Enum.UserInputType.Touch then
				local delta = input.Position - dragStart
				if not moved and delta.Magnitude > threshold then
					moved = true
					if onMove then onMove(true) end
				end
				if moved then
					local screen = gui.AbsoluteSize
					local newX = math.clamp(startPos.X.Offset + delta.X, 0, math.max(0, screen.X - target.AbsoluteSize.X))
					local newY = math.clamp(startPos.Y.Offset + delta.Y, 0, math.max(0, screen.Y - target.AbsoluteSize.Y))
					target.Position = UDim2.fromOffset(newX, newY)
				end
			end
		end))

		track(UIS.InputEnded:Connect(function(input)
			if not dragging then return end
			if input.UserInputType == Enum.UserInputType.MouseButton1
				or input.UserInputType == Enum.UserInputType.Touch then
				dragging = false
				if moved and onMove then onMove(false) end
				if not moved and onTap then
					onTap()
				end
			end
		end))
	end

	makeDraggable(titleBar, win, nil, 1, function(m)
		if L.borderTween and windowOpen then
			if m then L.borderTween:Pause() else L.borderTween:Play() end
		end
	end)
	makeDraggable(bubble, bubble, showWindow, 6)

	-- ====== UKURAN OTOMATIS (ikut layar: HP, tablet, PC, rotasi layar) ======
	function L.fit(instant)
		local v = gui.AbsoluteSize
		if v.X < 100 or v.Y < 100 then return end
		baseScale = L.calcScale(v)
		local w, h = math.floor(W * baseScale), math.floor(H * baseScale)
		win.Size = UDim2.fromOffset(w, h)
		if windowOpen then
			if instant then scaleObj.Scale = baseScale else tween(scaleObj, 0.25, {Scale = baseScale}) end
		end
		L.toastScale.Scale = math.clamp(baseScale, 0.75, 1.1)
		win.Position = UDim2.fromOffset(
			math.clamp(win.Position.X.Offset, 0, math.max(0, v.X - w)),
			math.clamp(win.Position.Y.Offset, 0, math.max(0, v.Y - h)))
		bubble.Position = UDim2.fromOffset(
			math.clamp(bubble.Position.X.Offset, 0, math.max(0, v.X - 48)),
			math.clamp(bubble.Position.Y.Offset, 0, math.max(0, v.Y - 48)))
	end
	track(gui:GetPropertyChangedSignal("AbsoluteSize"):Connect(function()
		task.delay(0.1, L.fit)   -- tunggu ukuran layar stabil (mis. saat rotasi)
	end))
	task.delay(0.8, L.fit)       -- sesuaikan sekali setelah animasi buka selesai

	function L.setScale(sc, instant)
		L.userScale = sc
		_G.ESP_SCALE = sc
		L.fit(instant)
		if not instant and L.refreshSizeBtns then L.refreshSizeBtns() end
	end

	do
		local grip = make("TextButton", {
			AnchorPoint = Vector2.new(1, 1), Position = UDim2.new(1, -2, 1, -2), Size = UDim2.fromOffset(24, 24),
			BackgroundTransparency = 1, Text = "", AutoButtonColor = false, ZIndex = 20,
		}, card)
		local dots = {}
		for _, d in ipairs({{16, 16}, {10, 16}, {16, 10}}) do
			local f = make("Frame", {
				Position = UDim2.fromOffset(d[1], d[2]), Size = UDim2.fromOffset(3, 3),
				BackgroundColor3 = THEME.muted, BackgroundTransparency = 0.45, BorderSizePixel = 0, ZIndex = 21,
			}, grip)
			corner(f, 2)
			table.insert(dots, f)
		end
		local function glow(a)
			for _, f in ipairs(dots) do tween(f, 0.15, {BackgroundTransparency = a}) end
		end
		local resizing, startPos, startScale = false, nil, 1
		grip.MouseEnter:Connect(function() if not resizing then glow(0.1) end end)
		grip.MouseLeave:Connect(function() if not resizing then glow(0.45) end end)
		grip.InputBegan:Connect(function(input)
			if input.UserInputType == Enum.UserInputType.MouseButton1
				or input.UserInputType == Enum.UserInputType.Touch then
				resizing = true
				startPos = input.Position
				startScale = baseScale
				if L.borderTween then L.borderTween:Pause() end
				glow(0)
			end
		end)
		track(UIS.InputChanged:Connect(function(input)
			if not resizing then return end
			if input.UserInputType ~= Enum.UserInputType.MouseMovement
				and input.UserInputType ~= Enum.UserInputType.Touch then return end
			local d = input.Position - startPos
			local v = gui.AbsoluteSize
			-- jangan melebihi tepi layar dari posisi jendela sekarang (biar tidak loncat)
			local cap = math.min((v.X - win.Position.X.Offset) / W, (v.Y - win.Position.Y.Offset) / H, 1.6)
			local sc = math.clamp(startScale + (d.X / W + d.Y / H) / 2, 0.4, math.max(0.4, cap))
			L.setScale(sc, true)
		end))
		track(UIS.InputEnded:Connect(function(input)
			if not resizing then return end
			if input.UserInputType == Enum.UserInputType.MouseButton1
				or input.UserInputType == Enum.UserInputType.Touch then
				resizing = false
				glow(0.45)
				if L.borderTween and windowOpen then L.borderTween:Play() end
				if L.refreshSizeBtns then L.refreshSizeBtns() end
			end
		end))
	end

	-- tombol E di keyboard
	track(UIS.InputBegan:Connect(function(input, gameProcessed)
		if gameProcessed then return end
		if input.KeyCode == Enum.KeyCode.E then
			toggle()
		elseif input.KeyCode == Enum.KeyCode.Q then
			if windowOpen then hideWindow() else showWindow() end
		elseif input.KeyCode == Enum.KeyCode.F then
			L.setFly(not L.fly.on)
		elseif input.KeyCode == Enum.KeyCode.N then
			L.setNoclip(not L.nc.on)
		elseif input.KeyCode == Enum.KeyCode.T then
			L.setTP(not L.tp.on)
		end
	end))

	local function tween(obj, time, props, style, dir)
		local t = TweenService:Create(obj, TweenInfo.new(time, style or Enum.EasingStyle.Quint, dir or Enum.EasingDirection.Out), props)

	local function addPress(button)
		local sc = make("UIScale", {Scale = 1}, button)
		local function release() tween(sc, 0.3, {Scale = 1}, Enum.EasingStyle.Back) end
		button.MouseButton1Down:Connect(function() tween(sc, 0.1, {Scale = 0.95}) end)
		button.MouseButton1Up:Connect(release)
		button.MouseLeave:Connect(release)
	end

	local function addHover(button, normal, hover)
		button.MouseEnter:Connect(function() tween(button, 0.18, {BackgroundColor3 = hover}) end)
		button.MouseLeave:Connect(function() tween(button, 0.25, {BackgroundColor3 = normal}) end)
	end

	end

	function L.T(k, ...)

		if select("#", ...) > 0 then return string.format(s, ...) end

	function L.bind(obj, prop, fn)
		obj[prop] = fn()
		table.insert(L.binds, {obj, prop, fn})
	end
	function L.refresh()
		for _, b in ipairs(L.binds) do
			if b[1].Parent then pcall(function() b[1][b[2]] = b[3]() end) end
		end
	end

	local function label(parent, props)
		local p = {
			BackgroundTransparency = 1,
			Font = Enum.Font.Gotham,
			TextSize = 12,
			TextColor3 = THEME.text,
			TextXAlignment = Enum.TextXAlignment.Left,
		}
		local key, fn = props.key, props.fn
		for k, v in pairs(props) do
			if k ~= "key" and k ~= "fn" then p[k] = v end
		end
		local o = make("TextLabel", p, parent)
		if key then L.bind(o, "Text", function() return L.T(key) end)
		elseif fn then L.bind(o, "Text", fn) end
		return o
	end

	-- kumpulkan semua target; dibagi per beberapa frame supaya tidak nge-lag
	local function gather(onlyOn)
		local list = {}
		local _, root = charParts()
		local cam = workspace.CurrentCamera
		local origin = root and root.Position or (cam and cam.CFrame.Position) or Vector3.zero
		local n = 0
		for _, o in ipairs(workspace:GetDescendants()) do
			for _, key in ipairs(CAT_ORDER) do
				if (not onlyOn or S.on[key]) and CATS[key].match(o) then
					local p = posOf(o)
					if p then list[#list + 1] = {obj = o, key = key, d = (p - origin).Magnitude} end
					break
				end
			end
			n += 1
			if n % 1500 == 0 then task.wait() end
		end
		table.sort(list, function(a, b) return a.d < b.d end)
		return list
	end

	-- ====== ESP ======
	local function clearAll()

	local function scan()
		local anyOn = false
		for _, k in ipairs(CAT_ORDER) do if S.on[k] then anyOn = true end end
		if not anyOn then
			clearAll()
			refreshStatus()
			return
		end
		if S.scanning then S.rescan = true return end
		S.scanning = true
		local list = gather(true)
		S.scanning = false
		if not S.alive then return end

		local want, cnt = {}, 0
		for _, e in ipairs(list) do
			if cnt >= MAX_ESP then break end
			if S.on[e.key] and e.obj.Parent then
				want[e.obj] = e.key
				cnt += 1
			end
		end

		for o, hl in pairs(S.active) do
			if not want[o] or not o.Parent or not hl.Parent then
				pcall(function() hl:Destroy() end)
				S.active[o] = nil
			end
		end
		for o, key in pairs(want) do
			local c = CATS[key].color
			local hl = S.active[o]
			if not (hl and hl.Parent) then
				hl = Instance.new("Highlight")
				hl.Name = "ESPSpeedHL"
				hl.FillTransparency = 1
				hl.OutlineTransparency = 1
				hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
				hl.Parent = o
				S.active[o] = hl
				tween(hl, 0.3, {FillTransparency = 0.35, OutlineTransparency = 0})
			end
			hl.FillColor = c
			hl.OutlineColor = c
		end
		refreshStatus()

		if S.rescan then
			S.rescan = false
			task.spawn(scan)
		end
	end

	local function flyStop()
		local f = S.fly
		if f.conn then f.conn:Disconnect() f.conn = nil end
		if f.bv then f.bv:Destroy() f.bv = nil end
		if f.bg then f.bg:Destroy() f.bg = nil end
		f.up, f.down = false, false
		local _, hrp, hum = charParts()
		if hum then hum.PlatformStand = false end
		if hrp then hrp.AssemblyLinearVelocity = Vector3.zero end
	end

	local function flyStart()
		flyStop()
		local _, hrp, hum = charParts()
		if not (hrp and hum) then return false end
		local f = S.fly
		local bv = Instance.new("BodyVelocity")
		bv.MaxForce = Vector3.new(9e9, 9e9, 9e9)
		bv.Velocity = Vector3.zero
		bv.Parent = hrp
		local bg = Instance.new("BodyGyro")
		bg.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
		bg.P = 9e4
		bg.D = 1000
		bg.CFrame = hrp.CFrame
		bg.Parent = hrp
		f.bv, f.bg = bv, bg
		hum.PlatformStand = true
		local cur = Vector3.zero
		f.conn = RunService.RenderStepped:Connect(function(dt)
			local cam = workspace.CurrentCamera
			if not cam or not hrp.Parent or not hum.Parent then return end
			local look = cam.CFrame.LookVector
			local flat = Vector3.new(look.X, 0, look.Z)
			flat = flat.Magnitude > 0.01 and flat.Unit or Vector3.new(0, 0, -1)
			local move = hum.MoveDirection              -- sudah relatif kamera (keyboard & joystick HP)
			local target = move * f.speed
			if move.Magnitude > 0.05 then               -- arah kamera naik/turun ikut menentukan ketinggian
				target = target + Vector3.new(0, look.Y * move:Dot(flat) * f.speed, 0)
			end
			local vy = 0
			if UIS:IsKeyDown(Enum.KeyCode.Space) or f.up then vy += 1 end
			if UIS:IsKeyDown(Enum.KeyCode.LeftShift) or UIS:IsKeyDown(Enum.KeyCode.LeftControl) or f.down then vy -= 1 end
			target = target + Vector3.new(0, vy * f.speed, 0)
			cur = cur:Lerp(target, math.clamp(dt * 10, 0, 1))   -- akselerasi halus
			bv.Velocity = cur
			bg.CFrame = CFrame.lookAt(hrp.Position, hrp.Position + flat)
		end)
		return true
	end

		local t = S.tps

			if not t.saved then
				t.saved = {player.CameraMode, player.CameraMinZoomDistance, player.CameraMaxZoomDistance}
			end
			pcall(function() player.CameraMode = Enum.CameraMode.Classic end)
			tpsZoom()
			if not t.conn then
				-- game kadang memaksa first person; tegakkan lagi tiap frame
				t.conn = RunService.RenderStepped:Connect(function()
					if not t.on then return end
					pcall(function()
						if player.CameraMode ~= Enum.CameraMode.Classic then
							player.CameraMode = Enum.CameraMode.Classic
							tpsZoom()
						end
						if not t.locking then
							if player.CameraMinZoomDistance > 0.5 then player.CameraMinZoomDistance = 0.5 end
							if player.CameraMaxZoomDistance < 20 then player.CameraMaxZoomDistance = 128 end
						end
						-- putar kamera dengan mouse: bebas saat menu ditutup, atau tahan klik kanan saat menu terbuka
						local cam = workspace.CurrentCamera
						if cam and cam.CameraType ~= Enum.CameraType.Custom then cam.CameraType = Enum.CameraType.Custom end
						if not UIS.TouchEnabled then
							local free = (not S.open) or UIS:IsMouseButtonPressed(Enum.UserInputType.MouseButton2)
							UIS.MouseBehavior = free and Enum.MouseBehavior.LockCenter or Enum.MouseBehavior.Default
							UIS.MouseIconEnabled = not free
						end
					end)
				end)
			end
		else
			if t.conn then t.conn:Disconnect() t.conn = nil end
			pcall(function()
				UIS.MouseBehavior = Enum.MouseBehavior.Default
				UIS.MouseIconEnabled = true
			end)
			if t.saved then
				local sv = t.saved
				t.saved = nil
				pcall(function()
					player.CameraMode = sv[1]
					player.CameraMinZoomDistance = sv[2]
					player.CameraMaxZoomDistance = sv[3]
				end)
			end
		end
		refreshStatus()
		if not silent then showToast(L.T(on and "toast_tps_on" or "toast_tps_off"), on) end
	end

	-- ====== TELEPORT ======
	local function getBounds(t)
		if t:IsA("Model") then
			local ok, cf, size = pcall(t.GetBoundingBox, t)
			if ok then return cf, size end
			return t:GetPivot(), Vector3.new(4, 4, 4)
		end
		return t.CFrame, t.Size
	end

	local function teleportTo(t)
		local ch, hrp = charParts()
		if not (ch and hrp) or not t:IsDescendantOf(workspace) then return false end
		local cf, size = getBounds(t)
		local center = cf.Position
		local floorY = center.Y - size.Y / 2
		local flat = Vector3.new(hrp.Position.X - center.X, 0, hrp.Position.Z - center.Z)
		local dir = flat.Magnitude > 0.5 and flat.Unit or Vector3.new(0, 0, 1)
		local radius = math.clamp(math.max(size.X, size.Z) / 2 + 4, 5, 16)
		local dest = Vector3.new(center.X, floorY + 3, center.Z) + dir * radius
		ch:PivotTo(CFrame.lookAt(dest, Vector3.new(center.X, dest.Y, center.Z)))
		hrp.AssemblyLinearVelocity = Vector3.zero
		return true
	end

	local function teleportToPlayer(p)
		local ch, hrp = charParts()
		local pch = p.Character
		local phrp = pch and pch:FindFirstChild("HumanoidRootPart")
		if not (ch and hrp and phrp) then return false end
		ch:PivotTo(phrp.CFrame * CFrame.new(0, 0, 5))
		hrp.AssemblyLinearVelocity = Vector3.zero
		return true
	end

	-- ====== DRAG ======
	local function makeDraggable(handle, target, onTap)
		local dragging, dragStart, startPos, moved = false, nil, nil, false
		track(handle.InputBegan:Connect(function(input)
			if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
				dragging = true
				moved = false
				dragStart = input.Position
				startPos = target.Position
			end
		end))
		track(UIS.InputChanged:Connect(function(input)
			if not dragging then return end
			if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
				local d = input.Position - dragStart
				if d.Magnitude > 6 then moved = true end
				if moved then
					local v = viewport()
					local sz = target.AbsoluteSize
					target.Position = UDim2.fromOffset(
						math.clamp(startPos.X.Offset + d.X, 0, math.max(0, v.X - sz.X)),
						math.clamp(startPos.Y.Offset + d.Y, 0, math.max(0, v.Y - sz.Y)))
				end
			end
		end))
		track(UIS.InputEnded:Connect(function(input)
			if not dragging then return end
			if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
				dragging = false
				if not moved and onTap then onTap() end
			end
		end))
	end

	local function setTab(i)

		S.tab = i

			tween(b, 0.25, {BackgroundTransparency = sel and 0 or 1, TextTransparency = sel and 0 or 0.45})
			local st = b:FindFirstChildOfClass("UIStroke")
			if sel and not st then subtleStroke(b) elseif not sel and st then st:Destroy() end
		end
		tween(UI.indicator, 0.4, {Position = UDim2.fromOffset(0, tabY(i) + 10)}, Enum.EasingStyle.Back)

		-- halaman lama: memudar sambil geser keluar
		for k, p in ipairs(UI.pages) do

				tween(p.group, 0.16, {
					GroupTransparency = 1,
					Position = UDim2.fromOffset(PAGE_POS.X.Offset, PAGE_POS.Y.Offset - dir * 12),
				}, Enum.EasingStyle.Quad)
			end
		end
		-- halaman baru: langsung muncul di saat yang sama
		local new = UI.pages[i].group
		if not new.Visible then
			new.Position = UDim2.fromOffset(PAGE_POS.X.Offset, PAGE_POS.Y.Offset + dir * 12)
			new.GroupTransparency = 1
			new.Visible = true
		end
		tween(new, 0.3, {GroupTransparency = 0, Position = PAGE_POS}, Enum.EasingStyle.Quint)

		task.delay(0.2, function()
			if my ~= S.tabToken then return end
			for k, p in ipairs(UI.pages) do
				if k ~= S.tab then p.group.Visible = false end
			end
		end)

		if i == 2 and UI.rebuild then UI.rebuild() end
	end

	end

	-- hanya teks yang memudar lalu muncul lagi dengan efek gelombang atas -> bawah
	local function animateSwap()
		local function collect()
			local out = {}
			for _, d in ipairs(card:GetDescendants()) do
				if (d:IsA("TextLabel") or d:IsA("TextButton")) and d.Text ~= "" and d.Text:find("%a")
					and not (d:IsDescendantOf(titleBar) and d ~= pillText) then
					local hidden = false
					for _, p in ipairs(UI.pages) do
						if not p.group.Visible and d:IsDescendantOf(p.group) then hidden = true break end
					end
					if not hidden then out[#out + 1] = d end
				end
			end
			return out
		end

		local old = collect()
		local goal = {}
		for _, d in ipairs(old) do
			goal[d] = d.TextTransparency
			tween(d, 0.12, {TextTransparency = 1}, Enum.EasingStyle.Quad)
		end
		task.delay(0.13, function()
			if not S.alive then return end
			applyLang()                                   -- teks diganti saat tak terlihat
			local seen = {}
			local top, hgt = card.AbsolutePosition.Y, math.max(1, card.AbsoluteSize.Y)
			for _, d in ipairs(collect()) do
				seen[d] = true
				local g = goal[d] or d.TextTransparency
				d.TextTransparency = 1
				local wave = math.clamp((d.AbsolutePosition.Y - top) / hgt, 0, 1) * 0.2
				task.delay(wave, function()
					if d.Parent then tween(d, 0.32, {TextTransparency = g}, Enum.EasingStyle.Quint) end
				end)
			end
			for _, d in ipairs(old) do                    -- yang tadi disembunyikan tapi tak ikut lagi: kembalikan
				if not seen[d] and d.Parent then d.TextTransparency = goal[d] end
			end
		end)
	end

	-- ====== SHUTDOWN ======
	local function shutdown()
		if not S.alive then return end
		S.alive = false
		pcall(setFly, false, true)
		pcall(setNoclip, false, true)
		pcall(setTPS, false, true)
		pcall(function()
			local _, _, h = charParts()
			if h and S.speed.orig then h.WalkSpeed = S.speed.orig end
		end)
		clearAll()
		for _, c in ipairs(conns) do pcall(function() c:Disconnect() end) end
		table.clear(conns)
		pcall(function() borderTween:Cancel() end)
		tween(card, 0.25, {GroupTransparency = 1})
		tween(borderStroke, 0.25, {Transparency = 1})
		tween(scaleObj, 0.3, {Scale = baseScale * 0.85})
		tween(bubbleScale, 0.2, {Scale = 0})
		task.delay(0.35, function() gui:Destroy() end)
		if _G.ESPSPEED_SHUTDOWN == shutdown then _G.ESPSPEED_SHUTDOWN = nil end
	end
